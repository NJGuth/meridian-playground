<aiir-coaching-guide>
  <metadata>
    <name>AIIR Leadership Coaching Guide</name>
    <purpose>Guidelines for interpreting assessment results and providing leadership coaching based on the AIIR Leadership Framework</purpose>
  </metadata>

  <philosophy>
    <core-belief>Leadership is a developmental process, not a destination. It can be improved throughout one's career through deliberate practice and attention to specific skills.</core-belief>
    
    <nature-vs-nurture>
      <insight>The "Great Man" theory has been eclipsed by theories that give much more weight to developmental forces.</insight>
      <insight>Leadership involves an interplay between innate characteristics and learned experience.</insight>
      <insight>The effort expended before current conditions is what in most cases allows for leadership to emerge.</insight>
      <quote author="Winston Churchill">"You have to learn from other people's mistakes — you don't have time to make them all yourself."</quote>
    </nature-vs-nurture>

    <skill-multiplicity>
      <principle>A narrow range of competencies is unlikely to lead to success in the modern corporate environment.</principle>
      <principle>Superior cognitive ability can be negated by inability to communicate and inspire others.</principle>
      <principle>"Spiky leadership" (excellence in few areas) must be accompanied by core competency in all areas.</principle>
      <catchphrase>The ideal executive is a jack of all trades and a master of some.</catchphrase>
    </skill-multiplicity>

    <organizational-pyramid>
      <insight>As leaders move up the organizational pyramid, they are less likely to apply specific trained skills and more likely to draw upon broader, more abstract capabilities.</insight>
      <insight>Executives exercise general problem-solving and social skills rather than narrow domain expertise.</insight>
    </organizational-pyramid>

  </philosophy>

  <aiir-method name="AIIR Method" description="A rigorous approach to coaching and leadership development proven to produce lasting change">
    <phase order="1" name="Assessment">
      <description>Extensive evaluation phase that typically includes psychometric assessments, 360° feedback, and developmental history interview.</description>
      <tools>
        <tool>LD12™ Assessment - assesses all domains, dimensions, and leadership skills</tool>
        <tool>360° Feedback</tool>
        <tool>Psychometric Assessments</tool>
        <tool>Developmental History Interview</tool>
      </tools>
    </phase>
    
    <phase order="2" name="Insight">
      <description>Coach works with the coachee to make meaning of the assessment data and derive understanding.</description>
      <activities>
        <activity>Review assessment results together</activity>
        <activity>Identify patterns and themes</activity>
        <activity>Connect data to real-world leadership experiences</activity>
        <activity>Develop shared understanding of strengths and growth areas</activity>
      </activities>
    </phase>
    
    <phase order="3" name="Implementation">
      <description>Setting data-driven development goals and creating action plans.</description>
      <activities>
        <activity>Set specific, measurable development goals</activity>
        <activity>Create action plans aligned with framework skills</activity>
        <activity>Focus coaching sessions on goal progression</activity>
        <activity>Apply learning in real leadership situations</activity>
      </activities>
    </phase>
    
    <phase order="4" name="Reinforcement">
      <description>Maintaining progress and embedding new behaviors beyond the coaching engagement.</description>
      <activities>
        <activity>Consistent touchpoints and check-ins</activity>
        <activity>Tips and goal-related content delivery</activity>
        <activity>Nudges to maintain progress</activity>
        <activity>Long-term behavior embedding</activity>
      </activities>
    </phase>
  </aiir-method>

  <assessment-interpretation>
    <profile-analysis>
      <balanced-profile>
        <description>A leader who exhibits consistently good performance across all skills, even if not exceptional in any single area.</description>
        <outcome>Will achieve superior results in the long run compared to "spiky" profiles.</outcome>
        <coaching-focus>Identify opportunities to elevate strong skills to exceptional while maintaining breadth.</coaching-focus>
      </balanced-profile>
      
      <spiky-profile>
        <description>A leader with some high-performing competencies offset by glaring gaps in other areas.</description>
        <outcome>Strengths may be negated by weaknesses; gains in strong areas offset by gaps.</outcome>
        <coaching-focus>Priority attention to addressing critical gaps while leveraging strengths.</coaching-focus>
      </spiky-profile>
    </profile-analysis>

    <performance-levels>
      <level name="High Performance" range="75-100">
        <description>Exceptional competency in this skill area</description>
        <coaching-approach>Leverage as strength; explore how to apply in supporting development of other skills</coaching-approach>
      </level>
      <level name="Good Performance" range="50-74">
        <description>Solid, reliable competency</description>
        <coaching-approach>Maintain and look for opportunities to elevate</coaching-approach>
      </level>
      <level name="Low Performance" range="0-49">
        <description>Gap area requiring development attention</description>
        <coaching-approach>Prioritize for development; create specific action plans</coaching-approach>
      </level>
    </performance-levels>

    <contextual-weighting>
      <principle>Different roles and conditions require different skill emphases.</principle>

      <context type="innovation-focus">
        <description>Roles requiring marketplace innovation</description>
        <priority-skills>Curiosity, Inspiration, Innovation, Navigating Change</priority-skills>
      </context>

      <context type="market-growth">
        <description>Roles focused on growing existing products/market share</description>
        <priority-skills>Managing Performance skills, Team Effectiveness skills</priority-skills>
      </context>

      <context type="culture-transformation">
        <description>Roles emphasizing culture change</description>
        <priority-skills>Trust, Authenticity, Culture Building, Inclusive Leadership</priority-skills>
      </context>

      <context type="turnaround">
        <description>Roles requiring organizational turnaround</description>
        <priority-skills>Resilience, Decision Quality, Navigating Change, Execution, Strategic Thinking</priority-skills>
      </context>

      <context type="scaling">
        <description>Roles focused on scaling teams and operations</description>
        <priority-skills>Delegation, Team Leadership, Developing Talent, Strategic Management</priority-skills>
      </context>
    </contextual-weighting>

  </assessment-interpretation>

  <coaching-guidance>
    <by-domain>
      <domain name="Leading Self">
        <focus>Foundation for all other leadership. Must be addressed before or alongside other domains.</focus>
        <common-development-areas>
          <area skill="Self-Awareness">Often the starting point—leaders must understand themselves before leading others effectively.</area>
          <area skill="Resilience">Critical for leaders facing ongoing challenges; often underdeveloped in high-achievers who haven't faced significant setbacks.</area>
          <area skill="Delegation">One of the least appreciated executive arts; many leaders struggle to let go.</area>
          <area skill="Emotion Management">Essential for maintaining composure in high-pressure situations.</area>
        </common-development-areas>
        <coaching-questions>
          <question>How do you currently attend to your mental, emotional, and physical well-being?</question>
          <question>What patterns do you notice in your emotional responses to challenging situations?</question>
          <question>How do you decide what to delegate versus what to handle yourself?</question>
          <question>When was the last time you took time for reflection away from day-to-day demands?</question>
        </coaching-questions>
      </domain>

      <domain name="Leading Others">
        <focus>Building relationships, developing team members, and creating high-performing teams.</focus>
        <common-development-areas>
          <area skill="Listening">Many leaders underestimate their tendency to interject opinions prematurely.</area>
          <area skill="Coaching">Shifting from "telling" to "asking" is a significant transition for many leaders.</area>
          <area skill="Empowering Others">Requires leaders to let go of control while maintaining accountability.</area>
          <area skill="Managing Conflict">Often avoided; constructive conflict management is a learnable skill.</area>
        </common-development-areas>
        <coaching-questions>
          <question>How do you ensure team members feel truly heard?</question>
          <question>What does empowering your team look like in practice?</question>
          <question>How do you approach conversations when there's disagreement?</question>
          <question>What investments are you making in developing your team members' capabilities?</question>
        </coaching-questions>
      </domain>

      <domain name="Leading the Organization">
        <focus>Strategic direction, organizational change, and culture. Most abstract but critically important.</focus>
        <common-development-areas>
          <area skill="Setting Vision">Requires articulating compelling future states, not just managing current operations.</area>
          <area skill="Strategic Thinking">Moving from tactical to strategic perspective is a major leadership transition.</area>
          <area skill="Culture Building">Often neglected; culture shouldn't be left to emerge spontaneously.</area>
          <area skill="Leadership Flexibility">Adapting style to situation requires self-awareness and range.</area>
        </common-development-areas>
        <coaching-questions>
          <question>What is the compelling vision you're working toward? How clear is it to others?</question>
          <question>How do you balance day-to-day operations with strategic thinking?</question>
          <question>What intentional actions are you taking to shape your organization's culture?</question>
          <question>How do you adapt your leadership approach to different situations?</question>
        </coaching-questions>
      </domain>
    </by-domain>

    <development-principles>
      <principle name="Start with Self-Awareness">Understanding oneself is prerequisite to developing other skills.</principle>
      <principle name="Address Critical Gaps First">Low-performance areas that impact effectiveness should be prioritized.</principle>
      <principle name="Leverage Strengths">Use high-performing skills to support development in other areas.</principle>
      <principle name="Context Matters">Weight skill importance based on current role demands and organizational needs.</principle>
      <principle name="Practice in Real Situations">Skills develop through application, not just understanding.</principle>
      <principle name="Seek Feedback">Regular feedback accelerates development and provides accountability.</principle>
      <principle name="Be Patient">Leadership development is a process measured in months and years, not days.</principle>
    </development-principles>

    <goal-setting>
      <smart-goals>
        <element name="Specific">Tied to specific skills in the framework</element>
        <element name="Measurable">Observable behaviors that can be tracked</element>
        <element name="Achievable">Realistic given current level and available support</element>
        <element name="Relevant">Connected to role requirements and organizational needs</element>
        <element name="Time-bound">Clear timeline for progress and review</element>
      </smart-goals>

      <example-goals>
        <goal skill="Delegation">
          <statement>Identify three significant projects to delegate to direct reports this quarter, providing clear outcomes and authority while maintaining weekly check-ins.</statement>
        </goal>
        <goal skill="Listening">
          <statement>In team meetings, practice asking clarifying questions before sharing my perspective, and solicit feedback from two team members on my listening effectiveness monthly.</statement>
        </goal>
        <goal skill="Strategic Thinking">
          <statement>Block two hours weekly for strategic reflection, and present a 3-year vision document to the leadership team within 60 days.</statement>
        </goal>
      </example-goals>
    </goal-setting>

  </coaching-guidance>

  <session-structure>
    <typical-coaching-session>
      <segment name="Check-in" duration="5-10 min">Review progress since last session, current state, and any urgent matters.</segment>
      <segment name="Goal Review" duration="10-15 min">Discuss progress on development goals, what's working, what's challenging.</segment>
      <segment name="Deep Dive" duration="20-30 min">Explore specific skill development, situations, or challenges in depth.</segment>
      <segment name="Action Planning" duration="10-15 min">Define specific actions and experiments for the coming period.</segment>
      <segment name="Wrap-up" duration="5 min">Summarize key insights and commitments.</segment>
    </typical-coaching-session>
  </session-structure>

  <response-guidelines>
    <when-assessing>
      <guideline>Reference specific skills and dimensions from the framework</guideline>
      <guideline>Consider the full profile, not just individual scores</guideline>
      <guideline>Account for role context and organizational needs</guideline>
      <guideline>Look for patterns across related skills</guideline>
      <guideline>Balance attention to gaps with recognition of strengths</guideline>
    </when-assessing>
    
    <when-coaching>
      <guideline>Ask powerful questions rather than providing answers</guideline>
      <guideline>Connect feedback to specific framework skills</guideline>
      <guideline>Help the leader see connections between skills</guideline>
      <guideline>Encourage real-world application and experimentation</guideline>
      <guideline>Support accountability without creating dependency</guideline>
    </when-coaching>
    
    <when-providing-feedback>
      <guideline>Be specific and behavioral</guideline>
      <guideline>Balance challenge with support</guideline>
      <guideline>Connect observations to framework language</guideline>
      <guideline>Offer multiple perspectives</guideline>
      <guideline>Encourage self-reflection</guideline>
    </when-providing-feedback>
  </response-guidelines>

      <typical-gaps>Delegation, Empowering Others, Influence, Executive Presence</typical-gaps>
      <development-focus>Shifting from individual contributor excellence to enabling others' success</development-focus>
    </profile>
    
    <profile name="People-Oriented Leader">
      <typical-strengths>Empathy, Building Relationships, Team Culture, Motivating Others</typical-strengths>
      <typical-gaps>Accountability, Execution, Decision Quality, Strategic Thinking</typical-gaps>
      <development-focus>Balancing relationship focus with results and strategic direction</development-focus>
    </profile>
    
    <profile name="Strategic Visionary">
      <typical-strengths>Setting Vision, Innovation, Strategic Thinking, Inspiration</typical-strengths>
      <typical-gaps>Execution, Time Management, Developing Talent, Prioritization</typical-gaps>
      <development-focus>Translating vision into reality through disciplined execution and team development</development-focus>
    </profile>
    
    <profile name="Results-Driven Operator">
      <typical-strengths>Execution, Accountability, Prioritization, Team Productivity</typical-strengths>
      <typical-gaps>Empathy, Listening, Innovation, Inclusive Leadership</typical-gaps>
      <development-focus>Developing people-centered leadership while maintaining operational excellence</development-focus>
    </profile>
  </example-leader-profiles>
